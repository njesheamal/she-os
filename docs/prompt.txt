## Pattern reference

**Repository — link:**
```python
def link_{entity_b}_to_{entity_a}(
    db: Session, {entity_a}_id: UUID, {entity_b}_id: UUID
) -> None:
    db.execute(
        {association_table}.insert().values(
            {entity_a}_id={entity_a}_id,
            {entity_b}_id={entity_b}_id,
        )
    )
    db.flush()
```

**Repository — unlink:**
```python
def unlink_{entity_b}_from_{entity_a}(
    db: Session, {entity_a}_id: UUID, {entity_b}_id: UUID
) -> bool:
    result = db.execute(
        delete({association_table}).where(
            {association_table}.c.{entity_a}_id == {entity_a}_id,
            {association_table}.c.{entity_b}_id == {entity_b}_id,
        )
    )
    db.flush()
    return result.rowcount > 0
```

**Router — link:**
```python
@router.post(
    "/{entity_a_id}/{entity_b_plural}/{entity_b_id}",
    status_code=status.HTTP_204_NO_CONTENT,
)
def link_{entity_a}_{entity_b}(
    {entity_a}_id: UUID,
    {entity_b}_id: UUID,
    db: Session = Depends(get_db),
):
    try:
        assoc_service.link_{entity_b}_to_{entity_a}(db, {entity_a}_id, {entity_b}_id)
    except {EntityA}NotFoundError as exc:
        raise HTTPException(status.HTTP_404_NOT_FOUND, str(exc)) from exc
    except {EntityB}NotFoundError as exc:
        raise HTTPException(status.HTTP_404_NOT_FOUND, str(exc)) from exc
    except AlreadyLinkedError as exc:
        raise HTTPException(status.HTTP_409_CONFLICT, str(exc)) from exc
```

**Router — unlink:**
```python
@router.delete(
    "/{entity_a_id}/{entity_b_plural}/{entity_b_id}",
    status_code=status.HTTP_204_NO_CONTENT,
)
def unlink_{entity_a}_{entity_b}(
    {entity_a}_id: UUID,
    {entity_b}_id: UUID,
    db: Session = Depends(get_db),
):
    try:
        assoc_service.unlink_{entity_b}_from_{entity_a}(db, {entity_a}_id, {entity_b}_id)
    except {EntityA}NotFoundError as exc:
        raise HTTPException(status.HTTP_404_NOT_FOUND, str(exc)) from exc
    except NotLinkedError as exc:
        raise HTTPException(status.HTTP_404_NOT_FOUND, str(exc)) from exc
```
